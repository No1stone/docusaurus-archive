# 작업 도커

```bash
sudo apt-get update -y && sudo apt-get upgrade -y
```

```bash
sudo apt install apt-transport-https ca-certificates curl software-properties-common -y
```

```bash
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /usr/share/keyrings/docker-archive-keyring.gpg
```

```bash
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null
```

```bash
sudo apt update
sudo apt install docker-ce docker-ce-cli containerd.io
```

```bash
sudo usermod -aG docker ${USER}
```

```bash
sudo systemctl start docker
sudo systemctl enable docker
```

```bash
sudo systemctl status docker

```

```bash
sudo curl -L "https://github.com/docker/compose/releases/download/v2.33.0/docker-compose-$(uname -s)-$(uname -m)" -o /usr/local/bin/docker-compose
```

```bash
# 현재 시스템에 docker 그룹 추가 (이미 있으면 무시됨)
sudo groupadd docker

# 모든 사용자(예: app2-user, app1-user, ubuntu 등)를 docker 그룹에 추가
sudo usermod -aG docker app2-user
sudo usermod -aG docker app1-user
sudo usermod -aG docker ubuntu
sudo usermod -aG docker data-user

sudo chmod 666 /var/run/docker.sock
```

```bash
# 현재 시스템에 docker 그룹 추가 (이미 있으면 무시됨)
sudo groupadd docker

# 모든 사용자(예: app2-user, app1-user, ubuntu 등)를 docker 그룹에 추가

sudo chmod 666 /var/run/docker.sock
```

```bash
sudo chmod +x /usr/local/bin/docker-compose

```

```bash
docker-compose --version
```
